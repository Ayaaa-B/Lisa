rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ==================================================
    // 1. OWNER AUTHENTICATION
    // ==================================================

    function isOwner() {
      return request.auth != null
        && request.auth.uid == "nails-by-lisa-owner";
    }


    // ==================================================
    // 2. BOOKING SETTINGS
    // ==================================================

    function maxPerDay() {
      let settingsPath =
        /databases/$(database)/documents/settings/main;

      return exists(settingsPath)
        && get(settingsPath).data.keys().hasAny(['maxPerDay'])
          ? get(settingsPath).data.maxPerDay
          : 3;
    }

    function validMaxPerDay(value) {
      return value is int
        && value >= 1
        && value <= 50;
    }

    function validSettings() {
      let d = request.resource.data;

      return d.keys().hasOnly([
          'maxPerDay',
          'daysAhead',
          'closedWeekdays'
        ])

        && (
          !d.keys().hasAny(['maxPerDay'])
          || validMaxPerDay(d.maxPerDay)
        )

        && (
          !d.keys().hasAny(['daysAhead'])
          || (
            d.daysAhead is int
            && d.daysAhead >= 0
            && d.daysAhead <= 180
          )
        )

        && (
          !d.keys().hasAny(['closedWeekdays'])
          || (
            d.closedWeekdays is list
            && d.closedWeekdays.size() <= 7
            && d.closedWeekdays.hasOnly([
              0, 1, 2, 3, 4, 5, 6
            ])
          )
        );
    }


    // ==================================================
    // 3. DATE VALIDATION
    // ==================================================

    function validDate(date) {
      return date is string
        && date.matches(
          '^[0-9]{4}-[0-9]{2}-[0-9]{2}$'
        );
    }


    // ==================================================
    // 4. NAIL SHAPE VALIDATION
    // ==================================================

    function validNailShape(shape) {
      return shape is string
        && shape in [
          'Almond',
          'Stiletto',
          'Duck',
          'Coffin'
        ];
    }


    // ==================================================
    // 5. MULTIPLE COLOUR VALIDATION
    // ==================================================

    function validColourValue(value) {
      return value is string
        && value.size() >= 1
        && value.size() <= 40;
    }

    function validColours(colours) {

      return colours is list

        && colours.size() >= 1

        && colours.size() <= 10

        && validColourValue(colours[0])

        && (
          colours.size() < 2
          || validColourValue(colours[1])
        )

        && (
          colours.size() < 3
          || validColourValue(colours[2])
        )

        && (
          colours.size() < 4
          || validColourValue(colours[3])
        )

        && (
          colours.size() < 5
          || validColourValue(colours[4])
        )

        && (
          colours.size() < 6
          || validColourValue(colours[5])
        )

        && (
          colours.size() < 7
          || validColourValue(colours[6])
        )

        && (
          colours.size() < 8
          || validColourValue(colours[7])
        )

        && (
          colours.size() < 9
          || validColourValue(colours[8])
        )

        && (
          colours.size() < 10
          || validColourValue(colours[9])
        );
    }


    // ==================================================
    // 6. CUSTOMER INSPIRATION PHOTO VALIDATION
    // ==================================================

    function validReferenceImage(bookingId) {

      let d = request.resource.data;

      return !d.keys().hasAny([
          'referenceImagePath'
        ])

        || (
          d.referenceImagePath is string

          && d.referenceImagePath.size() <= 500

          && d.referenceImagePath.matches(
            '^booking-references/'
            + bookingId
            + '/reference-[a-zA-Z0-9._-]+$'
          )
        );
    }


    // ==================================================
    // 7. BOOKING VALIDATION
    // ==================================================

    function validBooking(bookingId) {

      let d = request.resource.data;

      let dayPath =
        /databases/$(database)/documents/dayCounts/$(d.date);

      return bookingId.matches(
          '^[a-zA-Z0-9_-]{1,128}$'
        )

        && d.keys().hasAll([
          'date',
          'name',
          'phone',
          'shape',
          'colours',
          'style',
          'created'
        ])

        && d.keys().hasOnly([
          'date',
          'name',
          'phone',
          'shape',
          'colours',
          'style',
          'created',
          'referenceImagePath'
        ])


        // DATE

        && validDate(d.date)


        // CUSTOMER NAME

        && d.name is string
        && d.name.size() >= 2
        && d.name.size() <= 60


        // PHONE NUMBER

        && d.phone is string
        && d.phone.size() >= 7
        && d.phone.size() <= 25


        // NAIL SHAPE

        && validNailShape(d.shape)


        // MULTIPLE COLOURS

        && validColours(d.colours)


        // SERVICE

        && d.style is string
        && d.style.size() >= 1
        && d.style.size() <= 60


        // BOOKING TIMESTAMP

        && d.created is number


        // OPTIONAL CUSTOMER PHOTO

        && validReferenceImage(bookingId)


        // DAILY COUNT MUST BE UPDATED
        // IN THE SAME TRANSACTION

        && existsAfter(dayPath)

        && getAfter(dayPath).data.lastBookingId
          == bookingId

        && getAfter(dayPath).data.count is int

        && getAfter(dayPath).data.count >= 1

        && getAfter(dayPath).data.count <= maxPerDay()


        // DAILY COUNT MUST INCREASE BY EXACTLY ONE

        && (
          exists(dayPath)

          ? (
              get(dayPath).data.count is int

              && getAfter(dayPath).data.count
                == get(dayPath).data.count + 1
            )

          : getAfter(dayPath).data.count == 1
        );
    }


    // ==================================================
    // 8. OWNER DAILY COUNT VALIDATION
    // ==================================================

    function validOwnerDayCount() {

      let d = request.resource.data;

      return d.keys().hasOnly([
          'count',
          'lastBookingId'
        ])

        && d.count is int
        && d.count >= 0
        && d.count <= 100

        && (
          !d.keys().hasAny(['lastBookingId'])
          || d.lastBookingId is string
        );
    }


    // ==================================================
    // 9. CUSTOMER DAILY COUNT - NEW DOCUMENT
    // ==================================================

    function validNewBookingCount(date) {

      let bookingId =
        request.resource.data.lastBookingId;

      let bookingPath =
        /databases/$(database)/documents/bookings/$(bookingId);

      return validDate(date)

        && request.resource.data.keys().hasAll([
          'count',
          'lastBookingId'
        ])

        && request.resource.data.keys().hasOnly([
          'count',
          'lastBookingId'
        ])

        && request.resource.data.count is int

        && request.resource.data.count == 1

        && bookingId is string

        // The referenced booking must not already exist.

        && !exists(bookingPath)

        && existsAfter(bookingPath)

        && getAfter(bookingPath).data.date == date;
    }


    // ==================================================
    // 10. CUSTOMER DAILY COUNT - EXISTING DOCUMENT
    // ==================================================

    function validIncrementBookingCount(date) {

      let bookingId =
        request.resource.data.lastBookingId;

      let bookingPath =
        /databases/$(database)/documents/bookings/$(bookingId);

      return validDate(date)

        && request.resource.data.keys().hasAll([
          'count',
          'lastBookingId'
        ])

        && request.resource.data.keys().hasOnly([
          'count',
          'lastBookingId'
        ])

        && resource.data.count is int

        && request.resource.data.count is int


        // Customers can increase by exactly one.

        && request.resource.data.count
          == resource.data.count + 1

        && request.resource.data.count <= maxPerDay()

        && bookingId is string

        // The referenced booking must not already exist.

        && !exists(bookingPath)

        && existsAfter(bookingPath)

        && getAfter(bookingPath).data.date == date;
    }


    // ==================================================
    // 11. SETTINGS COLLECTION
    // ==================================================

    match /settings/{documentId} {

      allow read: if true;

      allow create, update: if isOwner()
        && (
          documentId != 'main'
          || validSettings()
        );

      allow delete: if isOwner();
    }


    // ==================================================
    // 12. COLOURS COLLECTION
    // ==================================================

    match /colours/{documentId} {

      allow read: if true;

      allow create, update, delete: if isOwner();
    }


    // ==================================================
    // 13. SERVICES COLLECTION
    // ==================================================

    match /styles/{documentId} {

      allow read: if true;

      allow create, update, delete: if isOwner();
    }


    // ==================================================
    // 14. GALLERY COLLECTION
    // ==================================================

    match /gallery/{documentId} {

      allow read: if true;

      allow create, update, delete: if isOwner();
    }


    // ==================================================
    // 15. DAILY BOOKING COUNTS
    // ==================================================

    match /dayCounts/{date} {

      // Customers can see available booking spaces.

      allow read: if true;


      // CREATE

      allow create: if
        (
          isOwner()
          && validOwnerDayCount()
        )
        ||
        validNewBookingCount(date);


      // UPDATE

      allow update: if
        (
          isOwner()
          && validOwnerDayCount()
        )
        ||
        validIncrementBookingCount(date);


      // DELETE

      allow delete: if isOwner();
    }


    // ==================================================
    // 16. BOOKINGS COLLECTION
    // ==================================================

    match /bookings/{bookingId} {

      // Customers can create valid bookings.

      allow create: if validBooking(bookingId);


      // Only Lisa can read, update or delete bookings.

      allow read, update, delete: if isOwner();
    }


    // ==================================================
    // 17. DENY EVERYTHING ELSE
    // ==================================================

    match /{document=**} {

      allow read, write: if false;
    }
  }
}
